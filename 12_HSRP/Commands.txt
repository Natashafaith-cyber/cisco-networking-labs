! 1. R1 Configuration (HSRPv2 Active Router with Priority & Preemption)
enable
configure terminal
 interface GigabitEthernet0/0
  standby version 2
  standby 1 ip 10.0.1.254
  standby 1 priority 150
  standby 1 preempt
 exit
end
write memory

! 2. R2 Configuration (HSRPv2 Standby Router with Lower Priority & Preempt)
enable
configure terminal
 interface GigabitEthernet0/0
  standby version 2
  standby 1 ip 10.0.1.254
  standby 1 priority 90
  standby 1 preempt
 exit
end
write memory

! 3. Verification Commands on R1 & R2
show standby
show standby brief
show ip route

! 4. Host Configuration & Failover Testing (PC1 & PC2)
! Set Default Gateway on PC1 and PC2 to: 10.0.1.254
! From PC Command Prompt:
ping 8.8.8.8
arp -a
