! 1. R1 Configuration (Enable IPv6 Routing & Interface IPv6 Addresses)
enable
configure terminal
 ipv6 unicast-routing

 interface GigabitEthernet0/0
  ipv6 address 2001:DB8:0:1::254/64
  no shutdown
 exit

 interface GigabitEthernet0/1
  ipv6 address 2001:DB8:0:2::254/64
  no shutdown
 exit

 interface GigabitEthernet0/2
  ipv6 address 2001:DB8:0:3::254/64
  no shutdown
 exit
end
write memory

! 2. Verification Commands on R1
show ipv6 interface brief
show ipv6 route
show ipv6 protocols

! 3. Host Configuration & Connectivity Tests (PC1, PC2, PC3)
 Configure each PC's IPv6 Address, Prefix Length (/64), and IPv6 Default Gateway
! in the PC Desktop -> IP Configuration panel:
   PC1: IPv6 Address = 2001:DB8:0:1::1/64 | Default Gateway = 2001:DB8:0:1::254
   PC2: IPv6 Address = 2001:DB8:0:2::1/64 | Default Gateway = 2001:DB8:0:2::254
   PC3: IPv6 Address = 2001:DB8:0:3::1/64 | Default Gateway = 2001:DB8:0:3::254

! Test IPv4 & IPv6 connectivity from PC Command Prompt:
ping 192.168.2.1
ping 192.168.3.1
ping 2001:DB8:0:2::1
ping 2001:DB8:0:3::1
