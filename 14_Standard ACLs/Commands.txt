! 1. OSPF Routing Configuration (R1 & R2)
! On R1:
enable
configure terminal
 router ospf 1
  router-id 1.1.1.1
  network 172.16.1.0 0.0.0.255 area 0
  network 172.16.2.0 0.0.0.255 area 0
  network 203.0.113.0 0.0.0.3 area 0
 exit
end

! On R2:
enable
configure terminal
 router ospf 1
  router-id 2.2.2.2
  network 192.168.1.0 0.0.0.255 area 0
  network 192.168.2.0 0.0.0.255 area 0
  network 203.0.113.0 0.0.0.3 area 0
 exit
end

! 2. Standard Numbered ACLs on R1 (Blocking 172.16.1.0/24 <-> 172.16.2.0/24)
! ACL 10 (outbound on G0/1 toward 172.16.2.0/24): block 172.16.1.0/24, allow others (e.g., 203.0.113.0/30)
! ACL 20 (outbound on G0/0 toward 172.16.1.0/24): block 172.16.2.0/24, allow others (e.g., 203.0.113.0/30)
enable
configure terminal
 access-list 10 deny 172.16.1.0 0.0.0.255
 access-list 10 permit any

access-list 20 deny 172.16.2.0 0.0.0.255
 access-list 20 permit any

 interface GigabitEthernet0/1
  ip access-group 10 out
 exit

 interface GigabitEthernet0/0
  ip access-group 20 out
 exit
end
write memory

! 3. Standard Named ACLs on R2 (Server Subnet Policies)
! ALLOW_192_168_1 (outbound on G0/0 toward 192.168.1.0/24): only PC1 (172.16.1.1) and PC3 (172.16.2.1)
! BLOCK_172_2_TO_192_2 (outbound on G0/1 toward 192.168.2.0/24): deny 172.16.2.0/24, permit others
enable
configure terminal
 ip access-list standard ALLOW_192_168_1
  permit host 172.16.1.1
  permit host 172.16.2.1
 exit
