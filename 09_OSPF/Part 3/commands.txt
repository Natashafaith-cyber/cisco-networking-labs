! 1. R1 & R2 Serial Link & OSPF Configuration
enable
configure terminal

! On the DCE router side (e.g., R1):
interface Serial0/0/0
 ip address 192.168.12.1 255.255.255.252
 clock rate 128000
 no shutdown
 ip ospf 1 area 0
exit

! On R2:
interface Serial0/0/0
 ip address 192.168.12.2 255.255.255.252
 no shutdown
 ip ospf 1 area 0
exit

! 2. Troubleshooting & Fixing OSPF Neighbor Adjacencies (e.g., R2, R4, R5)
enable
configure terminal
router ospf 1
 no passive-interface GigabitEthernet0/0
exit
interface GigabitEthernet0/0
 no shutdown
exit

! 3. Default Route & External Connectivity Fix (ASBR / Gateway)
enable
configure terminal
router ospf 1
 default-information originate
exit
ip route 0.0.0.0 0.0.0.0 203.0.113.1
end
write memory

! 4. Verification & Diagnostic Commands
show ip ospf neighbor
show ip route
show ip ospf database
show ip protocols
