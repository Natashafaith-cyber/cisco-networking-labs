! 1. Configuring Floating Static Routes on R1 & R2

! On R1 (Example configuration with a higher administrative distance of 130):
enable
configure terminal
ip route 10.0.2.0 255.255.255.0 <Next-Hop-IP> 130
end
write memory

! On R2:
enable
configure terminal
ip route 10.0.1.0 255.255.255.0 <Next-Hop-IP> 130
end
write memory

! 2. Testing & Link Failure Simulation Commands

! Simulate primary link failure for testing failover:
enable
configure terminal
interface GigabitEthernet0/2/0
 shutdown
end

! To restore the interface:
enable
configure terminal
interface GigabitEthernet0/2/0
 no shutdown
end

! 3. Verification Commands
show ip route
show ip route static
ping 10.0.2.1
ping 1.1.1.1
